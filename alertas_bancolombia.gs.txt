var URL   = 'https://rstcqffwrxrstwwfxrvj.supabase.co/rest/v1/rpc/alerta_banco?apikey=sb_publishable_m6tJglfiuQRBiFvrElqngQ_VHQi4jfF';
var TOKEN = 'PEGA_AQUI_EL_TOKEN';     // la app lo pone sola al usar "Copiar script"
// remitente real: alertasynotificaciones@an.notificacionesbancolombia.com
var BUSQUEDA = 'from:alertasynotificaciones newer_than:2d';
var CLAVE_ULTIMO = 'finanzas_ultimo_correo_ms';   // fecha del último correo procesado
var VERBOS = /(Compraste|Pagaste|Transferiste|Enviaste|Retiraste|Realizaste|Recibiste(?! este correo)|Te consignaron|Te transfirieron|Te enviaron)/i;

function procesarCorreos() {
  if (TOKEN === 'PEGA_AQUI_EL_' + 'TOKEN') throw new Error('Falta el TOKEN: copia el script desde la app (Ajustes → Correos del banco)');
  var props = PropertiesService.getScriptProperties();
  var ultimo = Number(props.getProperty(CLAVE_ULTIMO) || 0);
  if (!ultimo) ultimo = Date.now() - 24 * 60 * 60 * 1000;      // primera vez: las últimas 24 horas
  var maxVisto = ultimo, nuevos = 0, repetidos = 0, fallas = 0;
  // Gmail agrupa las alertas en hilos: se mira mensaje por mensaje y se compara por fecha.
  GmailApp.search(BUSQUEDA, 0, 30).forEach(function (hilo) {
    hilo.getMessages().forEach(function (msg) {
      var t = msg.getDate().getTime();
      if (t <= ultimo) return;
      var r = enviar_(extraer_(limpiar_(msg.getPlainBody() || msg.getBody())), 'mail-' + msg.getId(), msg.getDate());
      if (r && r.ok) { if (r.duplicado) repetidos++; else nuevos++; if (t > maxVisto) maxVisto = t; }
      else fallas++;
    });
  });
  if (!fallas) props.setProperty(CLAVE_ULTIMO, String(maxVisto));   // si algo falló, se reintenta en la próxima pasada
  if (!nuevos && !repetidos && !fallas) enviar_('', '', null);        // señal de vida para la app
  Logger.log('Nuevos: %s · repetidos: %s · fallas: %s', nuevos, repetidos, fallas);
}

// Deja solo la parte útil del correo: desde la frase del movimiento, máximo 700 letras.
function extraer_(t) {
  t = t.replace(/\s+/g, ' ').trim();
  var i = t.search(VERBOS);
  return (i >= 0 ? t.slice(Math.max(0, i - 30)) : t).slice(0, 700);
}

function limpiar_(html) {
  return String(html || '')
    .replace(/<style[\s\S]*?<\/style>/gi, ' ').replace(/<script[\s\S]*?<\/script>/gi, ' ')
    .replace(/<br\s*\/?>/gi, '\n').replace(/<\/p>/gi, '\n').replace(/<[^>]+>/g, ' ')
    .replace(/&nbsp;/g, ' ').replace(/&amp;/g, '&').replace(/&quot;/g, '"').replace(/&#39;/g, "'")
    .replace(/[ \t]+/g, ' ');
}

function enviar_(texto, uid, fecha) {
  var res = UrlFetchApp.fetch(URL, {
    method: 'post', contentType: 'application/json',
    payload: JSON.stringify({ p_token: TOKEN, p_texto: texto, p_uid: uid, p_fecha: fecha ? fecha.toISOString() : null }),
    muteHttpExceptions: true
  });
  try { return JSON.parse(res.getContentText()); } catch (e) { return null; }
}

// Prueba manual: muestra los correos del banco de las últimas 24 horas, SIN enviar nada.
function probar() {
  var desde = Date.now() - 24 * 60 * 60 * 1000, n = 0;
  GmailApp.search(BUSQUEDA, 0, 30).forEach(function (hilo) {
    hilo.getMessages().forEach(function (msg) {
      if (msg.getDate().getTime() <= desde) return;
      n++;
      Logger.log('→ ' + extraer_(limpiar_(msg.getPlainBody() || msg.getBody())).slice(0, 200));
    });
  });
  Logger.log('Correos del banco en las últimas 24 horas: ' + n + '. Si se ven bien, crea el activador cada 5 minutos para procesarCorreos.');
}

// Para volver a leer las últimas 24 horas: ejecutar esto y luego procesarCorreos.
function reiniciarMemoria() {
  PropertiesService.getScriptProperties().deleteProperty(CLAVE_ULTIMO);
  Logger.log('Memoria borrada: la próxima pasada toma las últimas 24 horas.');
}
